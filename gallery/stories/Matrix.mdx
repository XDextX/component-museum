export const contracts = import.meta.glob('../../contracts/*.contract.json', { eager: true });
export const components = Object.values(contracts).map((c) => (c.default ?? c).name);
export const matrix = frameworkMatrix({ react: reactEntry, astro: astroEntry }, components);

              <span aria-hidden="true">{matrix[name][fw] === 'present' ? '✓' : 'falta'}</span>
